-- VIEW: vw_city_service_gap
BEGIN;

CREATE OR REPLACE VIEW vw_city_service_gap AS
WITH months AS (
    SELECT DISTINCT make_date(year, month, 1) AS month_start FROM dim_date WHERE year IN (2023,
        2024)
), sales AS (
    SELECT date_trunc(&#x27;month&#x27;, sale_date)::date AS month_start, city, SUM(units_sold) AS units
    FROM fact_sales GROUP BY 1, 2
), complaints AS (
    SELECT date_trunc(&#x27;month&#x27;, complaint_date)::date AS month_start, city, COUNT(*) AS tickets
    FROM fact_complaints WHERE complaint_date IS NOT NULL GROUP BY 1, 2
), grid AS (
    SELECT m.month_start, c.city_name AS city, COALESCE(s.units, 0) AS monthly_sales,
            COALESCE(t.tickets, 0) AS monthly_complaints
    FROM months m CROSS JOIN dim_city c
    LEFT JOIN sales s ON s.month_start = m.month_start AND s.city = c.city_name
    LEFT JOIN complaints t ON t.month_start = m.month_start AND t.city = c.city_name
), fleet AS (
    SELECT *, SUM(monthly_sales) OVER (PARTITION BY city ORDER BY month_start ROWS UNBOUNDED
        PRECEDING) AS cumulative_vehicles
    FROM grid
), capacity AS (
    SELECT m.month_start, c.city, COUNT(*) AS active_centers,
            SUM(c.capacity_jobs_per_day) AS capacity_jobs_per_day
    FROM months m JOIN dim_service_center c
        ON c.opened_date &lt; m.month_start + INTERVAL &#x27;1 month&#x27; AND NOT c.source_record_missing
    GROUP BY m.month_start, c.city
)
SELECT f.*, COALESCE(c.active_centers, 0) AS active_centers, c.capacity_jobs_per_day,
        ROUND(f.cumulative_vehicles / NULLIF(c.active_centers, 0), 2) AS vehicles_per_center,
        ROUND(f.monthly_complaints * 1000.0 / NULLIF(f.cumulative_vehicles, 0), 2) AS
            complaints_per_1000_vehicles
FROM fleet f LEFT JOIN capacity c USING (month_start, city);

COMMIT;

-- VIEW: vw_monthly_kpis
BEGIN;

CREATE OR REPLACE VIEW vw_monthly_kpis AS
WITH fleet AS (
    SELECT month_start, SUM(monthly_sales) AS monthly_sales, SUM(cumulative_vehicles) AS
        cumulative_vehicles,
            SUM(active_centers) AS active_centers FROM vw_city_service_gap GROUP BY month_start
), complaints AS (
    SELECT date_trunc(&#x27;month&#x27;, complaint_date)::date AS month_start, COUNT(*) AS total_complaints,
            AVG(tat_days) AS avg_tat_days, COUNT(tat_days) AS known_tat_tickets,
            AVG(csat_score) FILTER (WHERE UPPER(TRIM(status)) IN (&#x27;RESOLVED&#x27;, &#x27;CLOSED&#x27;)) AS
                avg_csat_score,
            COUNT(csat_score) FILTER (WHERE UPPER(TRIM(status)) IN (&#x27;RESOLVED&#x27;, &#x27;CLOSED&#x27;)) AS
                csat_rated_resolved_tickets,
            COUNT(*) FILTER (WHERE UPPER(TRIM(status)) = &#x27;ESCALATED&#x27;) AS escalated_tickets
    FROM fact_complaints WHERE complaint_date IS NOT NULL GROUP BY 1
), registrations AS (
    SELECT date_trunc(&#x27;month&#x27;, reg_date)::date AS month_start, SUM(registrations) AS
        total_registrations,
            SUM(registrations) FILTER (WHERE LOWER(TRIM(oem)) = &#x27;ola electric&#x27;) AS ola_registrations
    FROM fact_vahan_registrations GROUP BY 1
), spares AS (
    SELECT date_trunc(&#x27;month&#x27;, week_start)::date AS month_start,
            CASE WHEN COUNT(stockout_days) = COUNT(*) THEN
                COUNT(*) FILTER (WHERE stockout_days &gt; 0) * 100.0 / NULLIF(COUNT(*), 0) END AS
                    stockout_rate_pct
    FROM fact_spare_parts GROUP BY 1
), social AS (
    SELECT date_trunc(&#x27;month&#x27;, feedback_date)::date AS month_start, COUNT(*) AS dated_social_posts,
            CASE WHEN COUNT(sentiment_label) = COUNT(*) THEN
                (COUNT(*) FILTER (WHERE sentiment_label = &#x27;Positive&#x27;) - COUNT(*) FILTER (WHERE
                    sentiment_label = &#x27;Negative&#x27;))::numeric
                / NULLIF(COUNT(*), 0) END AS sentiment_index
    FROM fact_sentiment WHERE source = &#x27;Social Media&#x27; AND feedback_date IS NOT NULL GROUP BY 1
), jobs AS (
    SELECT date_trunc(&#x27;month&#x27;, open_date)::date AS month_start,
            COUNT(*) FILTER (WHERE UPPER(TRIM(status)) = &#x27;AWAITING_PARTS&#x27;) * 100.0 /
                NULLIF(COUNT(*), 0) AS awaiting_parts_pct
    FROM fact_jobcards GROUP BY 1
)
SELECT EXTRACT(YEAR FROM f.month_start)::int AS year, EXTRACT(MONTH FROM f.month_start)::int AS
    month,
        f.month_start, f.monthly_sales, f.cumulative_vehicles, f.active_centers,
        COALESCE(c.total_complaints, 0) AS total_complaints,
        ROUND(c.avg_tat_days, 2) AS avg_tat_days, c.known_tat_tickets,
        ROUND(c.avg_csat_score, 2) AS avg_csat_score, c.csat_rated_resolved_tickets,
        ROUND(COALESCE(c.total_complaints, 0)*1000.0 / NULLIF(f.cumulative_vehicles, 0), 2) AS
            complaints_per_1000_vehicles,
        ROUND(f.cumulative_vehicles / NULLIF(f.active_centers, 0), 2) AS vehicles_per_center,
        ROUND(c.escalated_tickets*100.0 / NULLIF(c.total_complaints, 0), 2) AS escalation_rate_pct,
        ROUND(r.ola_registrations*100.0 / NULLIF(r.total_registrations, 0), 2) AS
            ola_market_share_pct,
        ROUND(s.stockout_rate_pct, 2) AS stockout_rate_pct,
        so.dated_social_posts, ROUND(so.sentiment_index, 4) AS sentiment_index,
        ROUND(j.awaiting_parts_pct, 2) AS awaiting_parts_pct
FROM fleet f LEFT JOIN complaints c USING(month_start) LEFT JOIN registrations r USING(month_start)
LEFT JOIN spares s USING(month_start) LEFT JOIN social so USING(month_start) LEFT JOIN jobs j
    USING(month_start);

COMMIT;

-- Reject duplicate source keys and ambiguous vehicle identifiers.
BEGIN;

DO $guard$
BEGIN
    IF EXISTS (SELECT 1 FROM fact_vahan_registrations WHERE oem &lt;&gt; UPPER(BTRIM(oem))) THEN
        RAISE EXCEPTION
            &#x27;Existing VAHAN OEM keys are noncanonical: migrate these keys in a separate transaction before running this script&#x27;;
    END IF;
END $guard$;
DO $check$
BEGIN
    IF EXISTS (SELECT 1 FROM _src_sales GROUP BY sale_date, city, model HAVING COUNT(*) &gt; 1) THEN
        RAISE EXCEPTION
            &#x27;Sales grain is not unique: sale_date, city, model; investigate before loading&#x27;;
    END IF;
    IF EXISTS (SELECT 1 FROM _src_complaints GROUP BY ticket_id HAVING COUNT(*) &gt; 1) THEN
        RAISE EXCEPTION &#x27;Duplicate cleaned ticket IDs; stop and review cleanup before loading&#x27;;
    END IF;
    IF EXISTS (SELECT 1 FROM public.master_id_map WHERE vehicle_reg_no IS NOT NULL
                GROUP BY vehicle_reg_no HAVING COUNT(DISTINCT chassis_no) &gt; 1) THEN
        RAISE EXCEPTION &#x27;A registration maps to several chassis; resolve identity before loading&#x27;;
    END IF;
    IF EXISTS (SELECT 1 FROM public.master_id_map
                WHERE COALESCE(customer_id, vehicle_reg_no, chassis_no, mobile_hash) IS NULL) THEN
        RAISE EXCEPTION &#x27;Identifier map contains an entirely unidentified row&#x27;;
    END IF;
    IF EXISTS (SELECT 1 FROM public.raw_spare_parts_consumption GROUP BY part_code
                HAVING COUNT(DISTINCT part_name) &gt; 1 OR COUNT(part_name) &lt; COUNT(*)) THEN
        RAISE EXCEPTION &#x27;Part codes have missing or conflicting names&#x27;;
    END IF;
END $check$;

DO $map$
BEGIN
    IF NOT EXISTS (SELECT 1 FROM public.master_id_map) THEN
        RAISE EXCEPTION &#x27;master_id_map is empty; do not load vehicle-dependent facts&#x27;;
    END IF;
    IF EXISTS (
        SELECT 1 FROM public.master_id_map
        GROUP BY customer_id, vehicle_reg_no, chassis_no, mobile_hash
        HAVING COUNT(*) &gt; 1
    ) THEN
        RAISE EXCEPTION &#x27;Duplicate map associations; review repeated map INSERTs before loading&#x27;;
    END IF;
END $map$;

COMMIT;

-- TABLE: _identity_map
BEGIN;

DROP TABLE IF EXISTS pg_temp._identity_map;

-- Reuse map associations; choose chassis before other identifiers.
CREATE TEMP TABLE _identity_map ON COMMIT PRESERVE ROWS AS
WITH registration_bridge AS (
    SELECT vehicle_reg_no, MIN(chassis_no) AS chassis_no
    FROM public.master_id_map WHERE vehicle_reg_no IS NOT NULL
    GROUP BY vehicle_reg_no
), enriched AS (
    SELECT m.*, COALESCE(m.chassis_no, b.chassis_no) AS canonical_chassis
    FROM public.master_id_map m
    LEFT JOIN registration_bridge b ON b.vehicle_reg_no = m.vehicle_reg_no
)
SELECT *, CASE WHEN canonical_chassis IS NOT NULL THEN &#x27;CHASSIS&#x27;
                WHEN vehicle_reg_no IS NOT NULL THEN &#x27;REGISTRATION&#x27;
                WHEN customer_id IS NOT NULL THEN &#x27;CUSTOMER_ONLY&#x27;
                ELSE &#x27;MOBILE_ONLY&#x27; END AS identity_type,
            COALESCE(canonical_chassis, vehicle_reg_no, customer_id, mobile_hash) AS identity_value
FROM enriched;

COMMIT;

-- TABLE: dim_date
BEGIN;

INSERT INTO dim_date (date_key, year, month, month_name, quarter, day_of_week)
WITH source_dates AS (
    SELECT DATE &#x27;2023-01-01&#x27; AS d UNION ALL SELECT DATE &#x27;2024-12-31&#x27;
    UNION ALL SELECT sale_date FROM _src_sales
    UNION ALL SELECT complaint_date FROM _src_complaints
    UNION ALL SELECT resolution_date FROM _src_complaints
    UNION ALL SELECT complaint_date_inferred FROM _src_complaints
    UNION ALL SELECT opened_date FROM public.raw_service_centers_master
    UNION ALL SELECT open_date FROM public.raw_job_cards_2024
    UNION ALL SELECT close_date FROM public.raw_job_cards_2024
    UNION ALL SELECT claim_date FROM public.raw_warranty_claims
    UNION ALL SELECT week_start FROM public.raw_spare_parts_consumption
    UNION ALL SELECT reg_date FROM public.raw_vahan_competitor_registrations
    UNION ALL SELECT review_date FROM public.raw_app_store_reviews
    UNION ALL SELECT post_timestamp_ist::DATE FROM public.raw_social_media_posts
    UNION ALL SELECT filed_date FROM public.raw_nch_ccpa_complaints
    UNION ALL SELECT incident_date FROM public.raw_competitor_incidents
), bounds AS (SELECT MIN(d) AS first_date, MAX(d) AS last_date FROM source_dates),
calendar AS (SELECT first_date + n AS datum FROM bounds,
                LATERAL generate_series(0, last_date - first_date) AS g(n))
SELECT datum, EXTRACT(YEAR FROM datum)::INT, EXTRACT(MONTH FROM datum)::INT,
        TRIM(TO_CHAR(datum, &#x27;Month&#x27;)), EXTRACT(QUARTER FROM datum)::INT,
        TRIM(TO_CHAR(datum, &#x27;Day&#x27;)) FROM calendar
ON CONFLICT (date_key) DO UPDATE SET
    year = EXCLUDED.year, month = EXCLUDED.month, month_name = EXCLUDED.month_name,
    quarter = EXCLUDED.quarter, day_of_week = EXCLUDED.day_of_week;

COMMIT;

BEGIN;

INSERT INTO fact_sentiment (
    source, source_id, feedback_date, rating, sentiment_label, classification_method, platform,
    feedback_text, posted_epoch_raw, likes, shares, comments
)
SELECT &#x27;App Store&#x27;, review_id, review_date, rating,
        CASE WHEN rating &gt;= 4 THEN &#x27;Positive&#x27; WHEN rating &lt;= 2 THEN &#x27;Negative&#x27;
            WHEN rating = 3 THEN &#x27;Neutral&#x27; END,
        &#x27;rating_rule&#x27;, platform, review_text, NULL::BIGINT, NULL::INT, NULL::INT, NULL::INT
FROM public.raw_app_store_reviews

SELECT &#x27;Social Media&#x27;, post_id, post_timestamp_ist::DATE, NULL::INT,
        CASE WHEN text IS NULL OR BTRIM(text) = &#x27;&#x27; THEN NULL
            WHEN text ~ *
                &#x27;(not good|not happy|not satisfied|bad|worst|poor|fail|broken|refund|delay|stranded|fire|smoke|unsafe|frustrat|disappoint|no response)&#x27;
                THEN &#x27;Negative&#x27;

        WHERE complaint_date IS NULL
    ) AS missing_complaint_dates,

                &#x27;OPEN&#x27;, &#x27;IN_PROGRESS&#x27;, &#x27;IN PROGRESS&#x27;, &#x27;ESCALATED&#x27;
